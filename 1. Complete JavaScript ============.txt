document.addEventListener('DOMContentLoaded', () => {

    // --------------------------------------
    // 1. MOBILE NAVIGATION TOGGLE
    // --------------------------------------
    const nav = document.querySelector('nav');
    const header = document.querySelector('header');
    
    // Create mobile menu button dynamically
    const menuToggle = document.createElement('button');
    menuToggle.classList.add('mobile-menu-toggle');
    menuToggle.setAttribute('aria-label', 'Toggle Navigation Menu');
    menuToggle.innerHTML = '&#9776; Menu';
    
    if (header && nav) {
        header.insertBefore(menuToggle, nav);
        menuToggle.addEventListener('click', () => {
            nav.classList.toggle('nav-open');
        });
    }

    // --------------------------------------
    // 2. CLIENT-SIDE FORM VALIDATION
    // --------------------------------------
    const enquiryForm = document.querySelector('.enquiry-form-section form');
    const contactForm = document.querySelector('.contact-form form');

    function validateForm(event) {
        event.preventDefault();
        const form = event.target;
        
        const nameInput = form.querySelector('input[type="text"]');
        const emailInput = form.querySelector('input[type="email"]');
        const messageInput = form.querySelector('textarea');

        let isValid = true;
        let errorMessage = '';

        // Validate Name
        if (!nameInput || nameInput.value.trim() === '') {
            isValid = false;
            errorMessage += '- Please enter your full name.\n';
        }

        // Validate Email format using Regex
        const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
        if (!emailInput || !emailPattern.test(emailInput.value.trim())) {
            isValid = false;
            errorMessage += '- Please enter a valid email address.\n';
        }

        // Validate Message
        if (!messageInput || messageInput.value.trim() === '') {
            isValid = false;
            errorMessage += '- Please write your message.\n';
        }

        // Display outcome
        if (!isValid) {
            alert('Form Errors:\n' + errorMessage);
        } else {
            alert('Thank you! Your enquiry has been received by Noir & Verse.');
            form.reset();
        }
    }

    if (enquiryForm) enquiryForm.addEventListener('submit', validateForm);
    if (contactForm) contactForm.addEventListener('submit', validateForm);

    // --------------------------------------
    // 3. INTERACTIVE PRODUCT FILTERING (products.html)
    // --------------------------------------
    const filterButtons = document.querySelectorAll('.filter-btn');
    const productItems = document.querySelectorAll('.product-item');

    filterButtons.forEach(button => {
        button.addEventListener('click', () => {
            const filterValue = button.getAttribute('data-filter');

            // Active button style toggle
            filterButtons.forEach(btn => btn.classList.remove('active'));
            button.classList.add('active');

            // Filter products display
            productItems.forEach(item => {
                if (filterValue === 'all' || item.classList.contains(filterValue)) {
                    item.style.display = 'block';
                } else {
                    item.style.display = 'none';
                }
            });
        });
    });

});

# Noir & Verse Apparel - Web Development Project

## Student Details
* **Module:** Web Development (WDFE5112 / WEDE5020)
* **Project Name:** Noir & Verse Apparel Prototype Website
* **Tech Stack:** HTML5, CSS3, JavaScript (Vanilla ES6), GitHub Pages

---

## Project Overview
Noir & Verse Apparel is an independent boutique clothing label blending vintage romance cinema, 35mm film photography, and written poetry into high-quality streetwear drops.

---

## Technical Features Implemented in Part 3

### 1. JavaScript Interactivity (`js/main.js`)
* **Client-Side Form Validation:** Prevents invalid form submissions on `enquiry.html` and `contact.html` by checking for empty input fields and valid email syntax via Regular Expressions.
* **Responsive Mobile Navigation:** Added a hamburger toggle button for seamless navigation on small mobile viewports.
* **Product Category Filter:** Implemented dynamic filtering buttons on `products.html` allowing users to view specific clothing items.

### 2. Search Engine Optimization (SEO) & Accessibility
* **Meta Tags:** Included `<meta name="description">`, `<meta name="keywords">`, and Open Graph social sharing tags across all 5 pages.
* **Semantic Structure:** Organized headings sequentially (`<h1>` through `<h4>`) to maintain content hierarchy.
* **Image Accessibility:** Assigned detailed `alt` text to every image tag for screen readers and SEO image search indexing.

### 3. Embedded External Services
* **Google Maps Embed:** Integrated an interactive map iframe on `contact.html` locating the brand's Cape Town pop-up store.
* **Spotify Playlist Embed:** Embedded an external music stream on `enquiry.html` to highlight the brand's mood boards and journal aesthetic.
